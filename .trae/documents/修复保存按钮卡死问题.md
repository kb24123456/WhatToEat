```swift
// 在 WhatToEatApp.swift 中添加初始数据填充功能
var sharedModelContainer: ModelContainer = {
    let schema = Schema([
        Restaurant.self,
        VisitLog.self,
    ])
    let modelConfiguration = ModelConfiguration(schema: schema, isStoredInMemoryOnly: false)

    do {
        let container = try ModelContainer(for: schema, configurations: [modelConfiguration])
        
        // 添加初始数据
        addInitialRestaurantData(to: container)
        
        return container
    } catch {
        fatalError("Could not create ModelContainer: \(error)")
    }
}()

private func addInitialRestaurantData(to container: ModelContainer) {
    let context = ModelContext(container)
    let descriptor = FetchDescriptor<Restaurant>()
    
    // 检查是否已有数据，避免重复添加
    if (try? context.fetch(descriptor))?.isEmpty ?? true {
        // 添加重庆市主城区真实餐厅数据
        let chongqingRestaurants = [
            Restaurant(
                name: "珮姐老火锅(解放碑店)",

                latitude: 29.5613,
                longitude: 106.5598,
                review: "正宗重庆老火锅，麻辣鲜香",
                tags: ["火锅", "重庆特色", "麻辣"],
                averagePrice: 120.0
            ),
            Restaurant(
                name: "周师兄火锅(洪崖洞店)",
                type: "火锅",
                district: "渝中",
                rating: 5,
                address: "重庆市渝中区嘉陵江滨江路88号",
                latitude: 29.5578,

                tags: ["火锅", "毛肚", "江景"],

            ),
            // 继续添加约50条类似格式的数据

    }
}
```
