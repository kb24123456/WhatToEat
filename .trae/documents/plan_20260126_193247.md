在 `AppTheme` 中添加二次编辑动效规范：
```swift
struct Animations {
    static let editingSpring = Animation.spring(response: 0.35, dampingFraction: 0.8)
    static let tagSpring = Animation.spring(response: 0.3, dampingFraction: 0.75)
    static let transitionScaleOpacity = .scale.combined(with: .opacity)
    static let transitionMoveTop = .move(edge: .top).combined(with: .opacity)
}
```
